\section{Settings of this data set}
\begin{itemize}
  \item \textbf{Ramp:} triangle of \SI{\RampFreq}{Hz} (10 periods in the capture) from
        \SIrange{\RampVmin}{\RampVmax}{V}; both end codes reached: \RampOverdriven.
  \item \textbf{Sines:} DDS amplitude setting \SI{\SineAmplMv}{mV} peak-to-peak, bias \SI{\SineBiasMv}{mV},
        odd prime number of periods in the capture of \AdcSamples{} samples.
  \item \textbf{ADC:} $f_s = \SI{\AdcFs}{MS/s}$; end-point line through $T_1$ and $T_{\AdcLastCode}$.
\end{itemize}
